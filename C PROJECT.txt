
 
#include <stdio.h>
#define MAX 50            /* maximum number of students */
int main(void)
{
    int rollNo[MAX];          /* roll numbers */
    char name[MAX][30];       /* student names */
    int m1[MAX], m2[MAX], m3[MAX];   /* marks in 3 subjects */
    int total[MAX];           /* total of 3 subjects */
    int count = 0;            /* students stored so far */
    int choice, i, key, found;
    do
    {
        /* show the menu */
        printf("\n===== STUDENT GRADE TRACKER =====\n");
        printf("1. Add student\n");
        printf("2. View all students\n");
        printf("3. Search student by roll number\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);
        switch (choice)
        {
        case 1:
            /* Add student */
            if (count == MAX)
            {
                printf("Array is full. Cannot add more students.\n");
                break;
            }
            printf("Enter roll number: ");

            scanf("%d", &rollNo[count]);
            printf("Enter name: ");
            scanf(" %29[^\n]", name[count]);
            printf("Enter marks in subject 1, 2 and 3 (0-100): ");
            scanf("%d %d %d", &m1[count], &m2[count], &m3[count]);
            if (m1[count] < 0 || m1[count] > 100 ||
                m2[count] < 0 || m2[count] > 100 ||
                m3[count] < 0 || m3[count] > 100)
            {
                printf("Invalid marks. Student not added.\n");
                break;
            }
            total[count] = m1[count] + m2[count] + m3[count];
            count++;
            printf("Student added. S.No. = %d\n", count);
            break;
        case 2:
            /* View all students with total */
            if (count == 0)
            {
                printf("No students added yet.\n");
                break;
            }
            printf("\n%-5s %-8s %-20s %-5s %-5s %-5s %-6s\n",
                   "S.No.", "Roll No", "Name", "Sub1", "Sub2", "Sub3", "Total");
            printf("-------------------------------------------------------------\n");
            for (i = 0; i < count; i++)
            {
                printf("%-5d %-8d %-20s %-5d %-5d %-5d %-6d\n",
                       i + 1, rollNo[i], name[i],

                       m1[i], m2[i], m3[i], total[i]);
            }
            break;
        case 3:
            /* Search student by roll number */
            if (count == 0)
            {
                printf("No students added yet.\n");
                break;
            }
            printf("Enter roll number to search: ");
            scanf("%d", &key);
            found = 0;
            for (i = 0; i < count; i++)
            {
                if (rollNo[i] == key)
                {
                    found = 1;
                    printf("\n%-5s %-8s %-20s %-5s %-5s %-5s %-6s\n",
                           "S.No.", "Roll No", "Name", "Sub1", "Sub2", "Sub3", "Total");
                    printf("-------------------------------------------------------------\n");
                    printf("%-5d %-8d %-20s %-5d %-5d %-5d %-6d\n",
                           i + 1, rollNo[i], name[i],
                           m1[i], m2[i], m3[i], total[i]);
                    break;
                }
            }
            if (found == 0)
            {
                printf("Student with roll number %d not found.\n", key);

            }
            break;
        case 4:
            printf("Thank you. Exiting...\n");
            break;
        default:
            printf("Invalid choice. Please enter 1 to 4.\n");
        }
    } while (choice != 4);
    return 0;
}
